Удержание когорт по дням (D1 / D7 / D30)
Формулы расчёта LTV
-- LTV = ARPU × Gross Margin × Lifetime SELECT user_id, AVG(order_value) AS arpu, 0.68 AS gross_margin, (1 / churn_rate) AS lifetime, arpu * gross_margin * (1/churn_rate) AS ltv FROM orders o JOIN users u ON o.user_id = u.id GROUP BY user_id 
-- Retention D1 / D7 / D30
SELECT 
    cohort_date, 
    COUNT_IF(days_since_first = 1) * 100.0 / COUNT(user_id) AS ret_d1, 
    COUNT_IF(days_since_first = 7) * 100.0 / COUNT(user_id) AS ret_d7, 
    COUNT_IF(days_since_first = 30) * 100.0 / COUNT(user_id) AS ret_d30 
FROM user_sessions 
GROUP BY cohort_date;
Логика данных · SQL & Data Quality
🗄️
SQL-запросы для KPI-метрик 
-- ROAS = Revenue / Ad Spend
SELECT 
    channel, 
    SUM(revenue) AS total_revenue, 
    SUM(ad_spend) AS total_spend, 
    SUM(revenue) / NULLIF(SUM(ad_spend), 0) AS roas 
FROM campaign_metrics 
WHERE date BETWEEN :start_date AND :end_date 
GROUP BY channel;
-- CAC = Ad Spend / New Customers
SELECT 
    c.channel, 
    SUM(c.ad_spend) / COUNT(DISTINCT u.user_id) AS cac 
FROM campaigns c 
JOIN users u 
    ON c.channel = u.acquisition_channel 
    AND u.is_new_customer = 1 
GROUP BY c.channel;
-- CPL = Spend / Leads
SELECT 
    channel, 
    campaign_id, 
    SUM(spend) / NULLIF(COUNT(lead_id), 0) AS cpl 
FROM ad_leads 
GROUP BY 1, 2 
ORDER BY cpl ASC;
-- CTR = Clicks / Impressions
SELECT 
    channel, 
    SUM(clicks) * 100.0 / NULLIF(SUM(impressions), 0) AS ctr_pct 
FROM ad_stats 
WHERE date >= CURRENT_DATE - INTERVAL '30 days' 
GROUP BY channel;
